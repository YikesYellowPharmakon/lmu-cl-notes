1. Pre-tokenization

BasePreTokenizer is the shared first stage for all tokenizers in this notebook. It takes a raw string and returns a flat list of string tokens — no vocabulary, no IDs, just the initial split and clean.

Implement normalize and pre_tokenize so that together they satisfy the following spec:

normalize(text)

  • Whitespace: Collapse all whitespace variants (tabs, newlines, Unicode spaces) to a plain space ' '
  • Case: Lowercase everything if self.lowercase = True, otherwise preserve case

pre_tokenize(text)

  • Split: Split the normalized string on whitespace only — whitespace is a boundary, never an output token
  • Punctuation: Strip all ASCII punctuation characters from each token after splitting
  • No empty strings: Stripping punctuation from a token like '!!' should produce nothing, not an empty string

import re, string

class BasePreTokenizer:
    """Normalizes and splits text into word-level chunks on whitespace."""
    def __init__(self, lowercase: bool = False):
        self.lowercase = lowercase

    def normalize(self, text: str) -> str:
        text = re.sub(r"\s+", " ", text)          # tabs, newlines, Unicode spaces -> ' '
        return text.lower() if self.lowercase else text

    def pre_tokenize(self, text: str) -> list[str]:
        table = str.maketrans("", "", string.punctuation)
        tokens = (tok.translate(table) for tok in text.split(" "))
        return [tok for tok in tokens if tok]

    def __call__(self, text: str) -> list[str]:
        return self.pre_tokenize(self.normalize(text))

p = BasePreTokenizer(lowercase=False)
print(p("Hello, World!")); print(p("hello!!")); print(p("foo.bar  baz!"))
print(BasePreTokenizer(lowercase=True)("Café Résumé"))
print(p("tab\there nbsp\nnew !! ."))
['Hello', 'World']
['hello']
['foobar', 'baz']
['café', 'résumé']
['tab', 'here', 'nbsp', 'new']

前四行就是卡片里的例子; 最后一行另外测了制表符、不换行空格 U+00A0、换行, 以及只由标点组成的「!!」「.」, 它们去掉标点后为空, 被过滤掉. normalize 之后连续空白已经合成一个空格, 所以 split(" ") 不会产生空串; 句首或句尾的空格会留下空串, 但最后一步的过滤也会把它去掉.

        """Adds new tokens to the vocabulary, skipping any that already exist."""
        for token in tokens:

                idx = len(self.token_to_id)
                self.token_to_id[token] = idx
                self.id_to_token[idx] = token

    def encode(self, text: str) -> list[int]:
        """Converts a string to a list of token IDs, using UNK for unknown tokens."""
        unk_id = self.token_to_id[self.UNK]
        return [self.token_to_id.get(token, unk_id) for token in self.pre_tokenizer(text)]

    def decode(self, ids: list[int]) -> str:
        """Converts a list of token IDs back to a space-joined string."""
        return " ".join(self.id_to_token.get(i, self.UNK) for i in ids)

    def save(self, path: str) -> None:
        """Saves the vocabulary to a JSON file."""
        with open(path, "w", encoding="utf-8") as f:
            json.dump({"vocab_size": self.vocab_size, "token_to_id": self.token_to_id}, f, indent=2)

    @classmethod
    def load(cls, path: str) -> "BaseTokenizer":
        """Loads a tokenizer from a saved JSON vocabulary file."""
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        tokenizer = cls(vocab_size=data["vocab_size"])
        tokenizer.token_to_id = data["token_to_id"]
        tokenizer.id_to_token = {int(i): t for t, i in data["token_to_id"].items()}
        return tokenizer

    def __call__(self, text: str) -> list[str]:
        return self.encode(text)

    def __str__(self) -> str:
        return f"BaseTokenizer(vocab_size={self.vocab_size}, loaded={len(self.token_to_id)})"

        """Applies learned BPE merges to a single word."""
        symbols = list(self._word_to_symbols(word))
        for pair in self.merges:

                    del symbols[i + 1]
                else:

        ids = []

            for sym in self._tokenize_word(word):
                ids.append(self.token_to_id.get(sym, unk_id))

        return "".join(tokens).replace(self.EOW, " ").strip()

def _get_pair_counts(self, vocab):
    counts = Counter()
    for symbols, freq in vocab.items():
        for a, b in zip(symbols, symbols[1:]):
            counts[(a, b)] += freq
    return counts

def _merge_pair(self, pair, vocab):
    """Replaces every adjacent occurrence of `pair` by the merged symbol."""
    a, b = pair
    new_vocab = {}
    for symbols, freq in vocab.items():
        out, i = [], 0
        while i < len(symbols):
            if i < len(symbols) - 1 and symbols[i] == a and symbols[i + 1] == b:
                out.append(a + b); i += 2
            else:
                out.append(symbols[i]); i += 1
        new_vocab[tuple(out)] = new_vocab.get(tuple(out), 0) + freq
    return new_vocab

def _best_pair(self, pair_counts):
    """Returns the most frequent pair, ties broken alphabetically."""
    return min(pair_counts.items(), key=lambda x: (-x[1], x[0]))[0]

def build_vocab(self, texts):
    word_counts = Counter(tok for text in texts for tok in self.pre_tokenizer(text))
    C = {tuple(w) + (self.EOW,): f for w, f in word_counts.items()}
    sym_freq = Counter()
    for symbols, f in C.items():
        for s in symbols:
            sym_freq[s] += f
    base = sorted(sym_freq)
    if len(base) >= self.vocab_size:
        print(f"warning: {len(base)} base symbols >= vocab_size, keeping the most frequent")
        base = sorted(sorted(base, key=lambda s: -sym_freq[s])[: self.vocab_size - 1])
    self.token_to_id = {self.UNK: 0}
    self.token_to_id.update({s: i + 1 for i, s in enumerate(base)})
    self.id_to_token = {i: t for t, i in self.token_to_id.items()}
    self.merges = []
    for _ in range(self.vocab_size - len(self.token_to_id)):
        pair_counts = self._get_pair_counts(C)
        if not pair_counts:
            break
        tl, tr = self._best_pair(pair_counts)
        C = self._merge_pair((tl, tr), C)
        self.merges.append((tl, tr))
        new = tl + tr
        if new not in self.token_to_id:
            j = len(self.token_to_id)
            self.token_to_id[new] = j
            self.id_to_token[j] = new

  3. 没见过的文字两种方法都救不了. 希腊语对所有配置都几乎全是 [UNK]: 词级 100%; BPE 约 81%, 只有 al-Ḵwārizmī、Muhammad 等拉丁字母部分能切出来, 每个希腊字母单独成为一个 [UNK] (所以每字符词元数接近 1). BPE 只能组合基本字母表里的符号, 字母表里没有的字符照样是未知的. 现代分词器用字节级 BPE (GPT-2 以来) 解决这个问题: 基本符号是 256 个字节, 任何文本都能编码, 不会有 [UNK].

  6. 大小写影响很小. 词级分词器保留大小写时英语 [UNK] 从 38.8% 升到 40.3% (多出来的那一个是句首的「Advanced」: 圣经里只有小写的 advanced); 对 BPE, 保留大小写使基本符号增加, 合并略少, 词元略多 (0.300 → 0.311), %UNK 几乎不变. 多语语料上词级分词器两种设置结果完全相同 (这几段文字里大写的词恰好都在或都不在词表里).

class MysteryTokenizer:
    def __init__(self, vocab_limit=1000, lowercase=True):
        self.vocab = set()
        self.vocab_limit = vocab_limit
        self.pre_tokenizer = BasePreTokenizer(lowercase=lowercase)

    def calculate_pair_scores(self, split_map, word_count):
        symbol_freq = defaultdict(int)
        pair_freq = defaultdict(int)

        for word, count in word_count.items():
            parts = split_map[word]
            for sym in parts:
                symbol_freq[sym] += count
            for a, b in zip(parts, parts[1:]):
                pair_freq[(a, b)] += count

        if not pair_freq:
            return {}

        return {
            pair: freq / (symbol_freq[pair[0]] * symbol_freq[pair[1]])
            for pair, freq in pair_freq.items()
        }

    def merge_best_pair(self, token_a, token_b, split_map):
        replacement = token_a + (token_b[2:] if token_b.startswith("??") else token_b)
        for word in split_map:
            parts = split_map[word]
            new = []
            i = 0
            while i < len(parts):
                if i < len(parts) - 1 and parts[i] == token_a and parts[i + 1] == token_b:
                    new.append(replacement)
                    i += 2
                else:
                    new.append(parts[i])
                    i += 1
            split_map[word] = new
        return split_map

    def build_vocabulary(self, texts: list[str]):
        word_count = Counter(token for text in texts for token in self.pre_tokenizer(text))

        split_map = {
            word: [word[0]] + [f"??{c}" for c in word[1:]]
            for word in word_count
        }

        base_symbols = sorted({sym for parts in split_map.values() for sym in parts})
        self.vocab = {"[PAD]", "[UNK]"} | set(base_symbols)

        while len(self.vocab) < self.vocab_limit:
            pair_scores = self.calculate_pair_scores(split_map, word_count)
            if not pair_scores:
                break

            best_pair = sorted(pair_scores.items(), key=lambda x: (-x[1], x[0]))[0][0]
            split_map = self.merge_best_pair(best_pair[0], best_pair[1], split_map)

            merged = best_pair[0] + (best_pair[1][2:] if best_pair[1].startswith("??") else best_pair[1])
            self.vocab.add(merged)

        print(f"Vocabulary size: {len(self.vocab)}")

    def save_vocabulary(self, path):
        with open(path, "w") as f:
            for token in sorted(self.vocab):
                f.write(token + "\n")

    def encode_single_word(self, word):
        tokens = []
        start = 0
        while start < len(word):
            end = len(word)
            match = None
            while start < end:
                substr = (f"??{word[start:end]}") if start > 0 else word[start:end]
                if substr in self.vocab:
                    match = substr
                    break
                end -= 1
            if match is None:
                return ["[UNK]"]
            tokens.append(match)
            start = end
        return tokens

    def tokenize_corpus(self, input_data):
        results = []
        for entry in input_data:
            words = self.pre_tokenizer(entry["sentence"])
            tokens = [tok for word in words for tok in self.encode_single_word(word)]
            results.append({"id": entry["id"], "tokens": tokens})
        return results

思路: 逐一比较三处: ① 初始切分: 词的非首字符带了什么前缀? ② 选哪一对合并: 打分公式是频率, 还是别的? ③ 编码: 是按合并规则重放, 还是在词表里找最长匹配? 遇到切不出来的词怎么办?

答案 (先自己做再看)

它和我们的 BPETokenizer 的三处不同:

  1. 初始切分和标记位置: BPE 在词尾加 </w>; MysteryTokenizer 给词的非首字符加前缀 ?? (「cat」→ c, ??a, ??t), 所以同一个字母在词首和词中是两个不同的符号. 这正是 BERT 的 ## 记号, 这里只是换成了 ??.
  2. 合并的打分: BPE 选出现次数最多的一对; 这里的分数是 score(a,b)=freq(a)⋅freq(b)freq(ab)​, 即这一对一起出现的次数相对于两个部件各自的频率. 两个部件都很常见时分数会被压低, 两个罕见部件总是一起出现时分数最高. 停止条件也不同: 不是固定的合并次数, 而是词表 (一个集合, 没有 ID) 达到 vocab_limit.
  3. 编码: BPE 按学到的合并顺序重放合并; 这里不用合并列表, 而是在词表里贪心找最长匹配: 从当前位置找能在词表里查到的最长子串, 切下来, 继续往后. 只要有一处查不到, 整个词就变成一个 [UNK] (BPE 只把查不到的那个字符变成 [UNK]).

所以这是 WordPiece (BERT、DistilBERT、ELECTRA 用的分词算法): 打分方式等价于「选能让训练数据似然提升最多的合并」, 编码用最长匹配优先 (MaxMatch).
